"""
Tests for a file upload feature (HTTP multipart endpoint).

Assumptions (adjust via environment variables or the constants below):
  - POST {BASE_URL}{UPLOAD_PATH} with multipart field "file"
  - Success -> 200/201 with JSON body containing an "id" or "url"
  - Rejections -> 400 (bad input), 413 (too large), 415 (bad type), 401/403 (auth)

Run:
  pip install pytest requests
  BASE_URL=http://localhost:8000 UPLOAD_PATH=/api/upload AUTH_TOKEN=xxx pytest -v test_file_upload.py
"""
import io
import os
import struct
import zlib

import pytest
import requests

BASE_URL = os.getenv("BASE_URL", "http://localhost:8000")
UPLOAD_PATH = os.getenv("UPLOAD_PATH", "/api/upload")
AUTH_TOKEN = os.getenv("AUTH_TOKEN", "")
FIELD_NAME = os.getenv("UPLOAD_FIELD", "file")

# Set these to match your server's configured limits / allowed types.
MAX_SIZE_BYTES = int(os.getenv("MAX_SIZE_BYTES", str(5 * 1024 * 1024)))  # 5 MB
ALLOWED_TYPES = {
    "image/png": ".png",
    "image/jpeg": ".jpg",
    "application/pdf": ".pdf",
    "text/plain": ".txt",
}

SUCCESS = (200, 201)


# ---------- helpers ----------

def url():
    return BASE_URL.rstrip("/") + UPLOAD_PATH


def headers(auth=True):
    return {"Authorization": f"Bearer {AUTH_TOKEN}"} if auth and AUTH_TOKEN else {}


def upload(filename, content, content_type, auth=True, field=FIELD_NAME, timeout=30):
    files = {field: (filename, io.BytesIO(content), content_type)}
    return requests.post(url(), files=files, headers=headers(auth), timeout=timeout)


def make_png(width=1, height=1):
    def chunk(tag, data):
        c = struct.pack(">I", len(data)) + tag + data
        return c + struct.pack(">I", zlib.crc32(tag + data) & 0xFFFFFFFF)

    raw = b"".join(b"\x00" + b"\xff\x00\x00" * width for _ in range(height))
    return (
        b"\x89PNG\r\n\x1a\n"
        + chunk(b"IHDR", struct.pack(">IIBBBBB", width, height, 8, 2, 0, 0, 0))
        + chunk(b"IDAT", zlib.compress(raw))
        + chunk(b"IEND", b"")
    )


PDF_BYTES = b"%PDF-1.4\n1 0 obj<</Type/Catalog>>endobj\ntrailer<</Root 1 0 R>>\n%%EOF\n"
TXT_BYTES = b"hello upload\n"


# ---------- happy path ----------

class TestSuccessfulUploads:
    def test_upload_png(self):
        r = upload("pic.png", make_png(), "image/png")
        assert r.status_code in SUCCESS, r.text
        body = r.json()
        assert any(k in body for k in ("id", "url", "path")), body

    def test_upload_pdf(self):
        r = upload("doc.pdf", PDF_BYTES, "application/pdf")
        assert r.status_code in SUCCESS, r.text

    def test_upload_text(self):
        r = upload("notes.txt", TXT_BYTES, "text/plain")
        assert r.status_code in SUCCESS, r.text

    def test_upload_at_exact_size_limit(self):
        r = upload("limit.txt", b"a" * MAX_SIZE_BYTES, "text/plain")
        assert r.status_code in SUCCESS, r.text

    def test_response_metadata_matches_upload(self):
        r = upload("meta.txt", TXT_BYTES, "text/plain")
        assert r.status_code in SUCCESS, r.text
        body = r.json()
        if "size" in body:
            assert body["size"] == len(TXT_BYTES)

    def test_duplicate_filename_does_not_overwrite(self):
        r1 = upload("dup.txt", b"first", "text/plain")
        r2 = upload("dup.txt", b"second", "text/plain")
        assert r1.status_code in SUCCESS and r2.status_code in SUCCESS
        j1, j2 = r1.json(), r2.json()
        ident = next((k for k in ("id", "url", "path") if k in j1), None)
        assert ident, j1
        assert j1[ident] != j2[ident], "Two uploads with the same name collided"

    def test_unicode_filename(self):
        r = upload("รายงาน_résumé_文档.txt", TXT_BYTES, "text/plain")
        assert r.status_code in SUCCESS, r.text


# ---------- validation: size ----------

class TestSizeValidation:
    def test_over_limit_rejected(self):
        r = upload("big.txt", b"a" * (MAX_SIZE_BYTES + 1), "text/plain")
        assert r.status_code == 413, r.status_code

    def test_empty_file_rejected(self):
        r = upload("empty.txt", b"", "text/plain")
        assert r.status_code in (400, 422), r.status_code


# ---------- validation: type ----------

class TestTypeValidation:
    @pytest.mark.parametrize(
        "name,ctype",
        [
            ("run.exe", "application/x-msdownload"),
            ("script.sh", "application/x-sh"),
            ("page.html", "text/html"),
            ("evil.svg", "image/svg+xml"),
        ],
    )
    def test_disallowed_types_rejected(self, name, ctype):
        r = upload(name, b"payload", ctype)
        assert r.status_code in (400, 415, 422), r.status_code

    def test_extension_content_type_mismatch(self):
        # Claims PNG, but bytes are a Windows executable header.
        r = upload("fake.png", b"MZ\x90\x00" + b"\x00" * 100, "image/png")
        assert r.status_code in (400, 415, 422), "Server trusted client-declared type"

    def test_double_extension_rejected(self):
        r = upload("photo.png.exe", make_png(), "image/png")
        assert r.status_code in (400, 415, 422), r.status_code

    def test_uppercase_extension_accepted(self):
        r = upload("PIC.PNG", make_png(), "image/png")
        assert r.status_code in SUCCESS, r.text


# ---------- malformed requests ----------

class TestMalformedRequests:
    def test_missing_file_field(self):
        r = requests.post(url(), data={"foo": "bar"}, headers=headers(), timeout=30)
        assert r.status_code in (400, 422), r.status_code

    def test_wrong_field_name(self):
        r = upload("a.txt", TXT_BYTES, "text/plain", field="not_the_file_field")
        assert r.status_code in (400, 422), r.status_code

    def test_non_multipart_body(self):
        r = requests.post(
            url(), data=b"raw bytes", headers={**headers(), "Content-Type": "text/plain"}, timeout=30
        )
        assert r.status_code in (400, 415, 422), r.status_code

    def test_missing_filename(self):
        r = upload("", TXT_BYTES, "text/plain")
        assert r.status_code in SUCCESS + (400, 422)  # either generated name or clean rejection


# ---------- security ----------

class TestSecurity:
    @pytest.mark.parametrize(
        "name",
        [
            "../../etc/passwd.txt",
            "..\\..\\windows\\system32\\x.txt",
            "/etc/cron.d/x.txt",
            "a/b/c.txt",
            "%2e%2e%2f%2e%2e%2fx.txt",
        ],
    )
    def test_path_traversal_filenames(self, name):
        r = upload(name, TXT_BYTES, "text/plain")
        if r.status_code in SUCCESS:
            stored = str(r.json())
            assert ".." not in stored and "etc/passwd" not in stored, stored
        else:
            assert r.status_code in (400, 422)

    def test_null_byte_in_filename(self):
        r = upload("safe.txt\x00.exe", TXT_BYTES, "text/plain")
        assert r.status_code in (400, 415, 422) or (
            r.status_code in SUCCESS and ".exe" not in str(r.json())
        )

    def test_very_long_filename(self):
        r = upload("a" * 1000 + ".txt", TXT_BYTES, "text/plain")
        assert r.status_code in SUCCESS + (400, 422)
        assert r.status_code < 500

    def test_script_in_filename_not_reflected_unescaped(self):
        name = '<script>alert(1)</script>.txt'
        r = upload(name, TXT_BYTES, "text/plain")
        assert r.status_code < 500
        assert "<script>" not in r.text

    def test_html_with_js_not_accepted_as_image(self):
        r = upload("x.png", b"<html><script>alert(1)</script></html>", "image/png")
        assert r.status_code in (400, 415, 422)

# ---------- robustness ----------

class TestRobustness:
    def test_multiple_files_in_one_request(self):
        files = [
            (FIELD_NAME, ("a.txt", io.BytesIO(b"a"), "text/plain")),
            (FIELD_NAME, ("b.txt", io.BytesIO(b"b"), "text/plain")),
        ]
        r = requests.post(url(), files=files, headers=headers(), timeout=30)
        # Either supported (success) or cleanly rejected; never a 5xx.
        assert r.status_code < 500

    def test_concurrent_uploads(self):
        from concurrent.futures import ThreadPoolExecutor

        def one(i):
            return upload(f"c{i}.txt", f"content {i}".encode(), "text/plain").status_code

        with ThreadPoolExecutor(max_workers=8) as ex:
            codes = list(ex.map(one, range(16)))
        assert all(c in SUCCESS for c in codes), codes

    def test_aborted_upload_does_not_crash_server(self):
        import socket
        from urllib.parse import urlparse

        p = urlparse(url())
        s = socket.create_connection((p.hostname, p.port or 80), timeout=5)
        s.sendall(
            f"POST {p.path} HTTP/1.1\r\nHost: {p.hostname}\r\n"
            f"Content-Type: multipart/form-data; boundary=x\r\nContent-Length: 100000\r\n\r\n--x\r\n".encode()
        )
        s.close()
        # Server should still respond afterwards.
        r = upload("after.txt", TXT_BYTES, "text/plain")
        assert r.status_code in SUCCESS
